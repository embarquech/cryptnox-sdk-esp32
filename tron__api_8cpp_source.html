59static void wifi_event_handler(void *arg, esp_event_base_t event_base,
60 int32_t event_id, void *event_data)
61{
62 (void)arg;
63 (void)event_data;
64
65 if ((event_base == WIFI_EVENT) && (event_id == WIFI_EVENT_STA_START)) {
66 esp_wifi_connect();
67 } else if ((event_base == WIFI_EVENT) &&
68 (event_id == WIFI_EVENT_STA_DISCONNECTED)) {
69 if (s_retry_num < WIFI_MAX_RETRY) {
70 esp_wifi_connect();
71 s_retry_num++;
72 ESP_LOGW(TAG, "WiFi retry %d/%d", s_retry_num, WIFI_MAX_RETRY);
73 } else {
74 xEventGroupSetBits(s_wifi_event_group, WIFI_FAIL_BIT);
75 }
76 } else if ((event_base == IP_EVENT) && (event_id == IP_EVENT_STA_GOT_IP)) {
77 s_retry_num = 0;
78 xEventGroupSetBits(s_wifi_event_group, WIFI_CONNECTED_BIT);
79 } else {
80 /* other events ignored */
81 }
82}

84/******************************************************************
85 * HTTP helper
86 ******************************************************************/
87
88/* A response is only usable with a 2xx status. */
89#define HTTP_STATUS_SUCCESS_MIN (200)
90#define HTTP_STATUS_SUCCESS_MAX (299)
91
92/*
93 * POST 'body' to s_base_url + path over HTTPS and read the response into
94 * resp_buf (NUL-terminated). Returns true only for a complete 2xx body that
95 * fit in resp_buf — a truncated, errored, or non-2xx read is a failure.
96 */
97static bool do_post(const char *path, const char *body,
98 char *resp_buf, size_t resp_buf_size)
99{
100 bool success = false;
101
102 char url[URL_BUF_SIZE];
103 if ((size_t)snprintf(url, sizeof(url), "%s%s", s_base_url, path) >= sizeof(url)) {
104 ESP_LOGE(TAG, "URL too long");
105 return false;
106 }
107
108 esp_http_client_config_t cfg;
109 (void)memset(&cfg, 0, sizeof(cfg));
110 cfg.url = url;
111 cfg.method = HTTP_METHOD_POST;
112 cfg.timeout_ms = 15000;
113 cfg.crt_bundle_attach = esp_crt_bundle_attach;
114
115 esp_http_client_handle_t client = esp_http_client_init(&cfg);
116 if (client == NULL) {
117 ESP_LOGE(TAG, "HTTP client init failed");
118 return false;
119 }
120
121 (void)esp_http_client_set_header(client, "Content-Type", "application/json");
122 if ((s_api_key != NULL) && (s_api_key[0] != '\0')) {
123 (void)esp_http_client_set_header(client, "TRON-PRO-API-KEY", s_api_key);
124 }
125
126 int body_len = (int)strlen(body);
127 esp_err_t err = esp_http_client_open(client, body_len);
128 if (err != ESP_OK) {
129 ESP_LOGE(TAG, "HTTP open: %s", esp_err_to_name(err));
130 goto cleanup;
131 }
132
133 if (esp_http_client_write(client, body, body_len) != body_len) {
134 ESP_LOGE(TAG, "HTTP write incomplete");
135 goto cleanup;
136 }
137
138 {
139 (void)esp_http_client_fetch_headers(client); /* may be -1 when chunked */
140
141 int total = 0;
142 int read = 0;
143 do {
144 int space = (int)(resp_buf_size - 1U) - total;
145 if (space <= 0) {
146 read = 0; /* buffer full: completeness is checked below */
147 } else {
148 read = esp_http_client_read(client, resp_buf + total, space);
149 if (read > 0) {
150 total += read;
151 }
152 }
153 } while (read > 0);
154
155 resp_buf[total] = '\0';
156
157 int status = esp_http_client_get_status_code(client);
158 if (read < 0) {
159 ESP_LOGE(TAG, "HTTP read error after %d bytes", total);
160 } else if (!esp_http_client_is_complete_data_received(client)) {
161 ESP_LOGE(TAG, "HTTP response truncated at %d bytes", total);
162 } else if ((status < HTTP_STATUS_SUCCESS_MIN) || (status > HTTP_STATUS_SUCCESS_MAX)) {
163 ESP_LOGE(TAG, "HTTP status %d: %s", status, resp_buf);
164 } else {
165 success = (total > 0);
166 }
167 }
168
169cleanup:
170 esp_http_client_close(client);
171 esp_http_client_cleanup(client);
172 return success;
173}

185bool tron_api_wifi_connect(const char *ssid, const char *password)
186{
187 s_wifi_event_group = xEventGroupCreate();
188 s_retry_num = 0;
189
190 ESP_ERROR_CHECK(esp_netif_init());
191 ESP_ERROR_CHECK(esp_event_loop_create_default());
192 (void)esp_netif_create_default_wifi_sta();
193
194 wifi_init_config_t cfg = WIFI_INIT_CONFIG_DEFAULT();
195 ESP_ERROR_CHECK(esp_wifi_init(&cfg));
196
197 esp_event_handler_instance_t h_any;
198 esp_event_handler_instance_t h_ip;
199 ESP_ERROR_CHECK(esp_event_handler_instance_register(
200 WIFI_EVENT, ESP_EVENT_ANY_ID, &wifi_event_handler, NULL, &h_any));
201 ESP_ERROR_CHECK(esp_event_handler_instance_register(
202 IP_EVENT, IP_EVENT_STA_GOT_IP, &wifi_event_handler, NULL, &h_ip));
203
204 wifi_config_t wifi_cfg;
205 (void)memset(&wifi_cfg, 0, sizeof(wifi_cfg));
206 (void)strncpy((char *)wifi_cfg.sta.ssid, ssid, sizeof(wifi_cfg.sta.ssid) - 1U);
207 (void)strncpy((char *)wifi_cfg.sta.password, password, sizeof(wifi_cfg.sta.password) - 1U);
208 wifi_cfg.sta.threshold.authmode = WIFI_AUTH_WPA2_PSK;
209
210 ESP_ERROR_CHECK(esp_wifi_set_mode(WIFI_MODE_STA));
211 ESP_ERROR_CHECK(esp_wifi_set_config(WIFI_IF_STA, &wifi_cfg));
212 ESP_ERROR_CHECK(esp_wifi_start());
213
214 ESP_LOGI(TAG, "Connecting to \"%s\"...", ssid);
215
216 EventBits_t bits = xEventGroupWaitBits(s_wifi_event_group,
217 WIFI_CONNECTED_BIT | WIFI_FAIL_BIT,
218 pdFALSE, pdFALSE,
219 pdMS_TO_TICKS(WIFI_TIMEOUT_MS));
220
221 bool connected = ((bits & WIFI_CONNECTED_BIT) != 0U);
222 if (connected) {
223 ESP_LOGI(TAG, "WiFi connected");
224 } else {
225 ESP_LOGE(TAG, "WiFi connect failed");
226 }
227
228 (void)esp_event_handler_instance_unregister(IP_EVENT, IP_EVENT_STA_GOT_IP, h_ip);
229 (void)esp_event_handler_instance_unregister(WIFI_EVENT, ESP_EVENT_ANY_ID, h_any);
230 vEventGroupDelete(s_wifi_event_group);
231
232 return connected;
233}

235bool tron_api_create_transfer(const char *owner_b58, const char *to_b58,
236 uint64_t amount,
237 char *tx_json, size_t tx_json_size)
238{
239 char body[CREATE_BODY_SIZE];
240 (void)snprintf(body, sizeof(body),
241 "{\"owner_address\":\"%s\",\"to_address\":\"%s\","
242 "\"amount\":%" PRIu64 ",\"visible\":true}",
243 owner_b58, to_b58, amount);
244
245 if (!do_post("/wallet/createtransaction", body, tx_json, tx_json_size)) {
246 return false;
247 }
248
249 /* The node reports build errors as {"Error":"..."} with HTTP 200. */
250 if (strstr(tx_json, "\"txID\"") == NULL) {
251 ESP_LOGE(TAG, "createtransaction failed: %s", tx_json);
252 return false;
253 }
254
255 return true;
256}

258bool tron_api_json_string(const char *json, const char *key,
259 char *out, size_t out_size)
260{
261 char pattern[32];
262 if ((size_t)snprintf(pattern, sizeof(pattern), "\"%s\":\"", key) >= sizeof(pattern)) {
263 return false;
264 }
265
266 const char *start = strstr(json, pattern);
267 if (start == NULL) {
268 return false;
269 }
270 start += strlen(pattern);
271
272 const char *end = strchr(start, '"');
273 if (end == NULL) {
274 return false;
275 }
276
277 size_t len = (size_t)(end - start);
278 if ((len + 1U) > out_size) {
279 return false;
280 }
281
282 (void)memcpy(out, start, len);
283 out[len] = '\0';
284 return true;
285}

287bool tron_api_hex_to_bytes(const char *hex, uint8_t *out, size_t out_size,
288 size_t *out_len)
289{
290 size_t hex_len = strlen(hex);
291 if (((hex_len % HEX_PER_BYTE) != 0U) || ((hex_len / HEX_PER_BYTE) > out_size)) {
292 return false;
293 }
294
295 size_t i;
296 for (i = 0U; i < (hex_len / HEX_PER_BYTE); i++) {
297 uint8_t byte = 0U;
298 for (size_t n = 0U; n < HEX_PER_BYTE; n++) {
299 char c = hex[(i * HEX_PER_BYTE) + n];
300 uint8_t nib;
301 if ((c >= '0') && (c <= '9')) { nib = (uint8_t)(c - '0'); }
302 else if ((c >= 'a') && (c <= 'f')) { nib = (uint8_t)(c - 'a' + 10); }
303 else if ((c >= 'A') && (c <= 'F')) { nib = (uint8_t)(c - 'A' + 10); }
304 else { return false; }
305 byte = (uint8_t)((byte << 4U) | nib);
306 }
307 out[i] = byte;
308 }
309
310 *out_len = i;
311 return true;
312}

314bool tron_api_broadcast(const char *tx_json,
315 const uint8_t signature[TRON_SIGNATURE_BYTES],
316 char *resp, size_t resp_size)
317{
318 static const char HEX[] = "0123456789abcdef";
319
320 /* Never leave the caller's buffer unwritten — it gets logged on failure. */
321 if ((resp == NULL) || (resp_size == 0U)) {
322 return false;
323 }
324 resp[0] = '\0';
325
326 if (tx_json[0] != '{') {
327 return false;
328 }
329
330 /* Splice "signature":["<hex>"], in right after the opening brace so the
331 * node receives back exactly the transaction object it produced. */
332 char sig_field[SIG_FIELD_SIZE];
333 size_t p = 0U;
334 const char *prefix = "\"signature\":[\"";
335 (void)memcpy(&sig_field[p], prefix, strlen(prefix));
336 p += strlen(prefix);
337 for (size_t i = 0U; i < TRON_SIGNATURE_BYTES; i++) {
338 sig_field[p++] = HEX[(signature[i] >> 4U) & 0x0FU];
339 sig_field[p++] = HEX[signature[i] & 0x0FU];
340 }
341 sig_field[p++] = '"';
342 sig_field[p++] = ']';
343 sig_field[p++] = ',';
344 sig_field[p] = '\0';
345
346 size_t body_size = strlen(tx_json) + p + 2U;
347 char *body = (char *)malloc(body_size);
348 if (body == NULL) {
349 return false;
350 }
351 (void)snprintf(body, body_size, "{%s%s", sig_field, &tx_json[1]);
352
353 bool posted = do_post("/wallet/broadcasttransaction", body, resp, resp_size);
354 free(body);
355
356 if (!posted) {
357 return false;
358 }
359
360 return (strstr(resp, "\"result\":true") != NULL);
361}

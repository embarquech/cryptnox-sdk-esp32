120static bool verify_transaction(const char *tx_json, CW_CryptoProvider &crypto,
121 const uint8_t *owner21, const uint8_t *to21,
122 uint64_t amount, uint8_t *hash)
123{
124 char raw_hex[HEX_STR_MAX];
125 char txid_hex[(CW_HASH_SIZE * 2U) + 1U];
126
127 if (!tron_api_json_string(tx_json, "raw_data_hex", raw_hex, sizeof(raw_hex)) ||
128 !tron_api_json_string(tx_json, "txID", txid_hex, sizeof(txid_hex))) {
129 ESP_LOGE(TAG, "transaction is missing raw_data_hex / txID");
130 return false;
131 }
132
133 uint8_t raw[RAW_DATA_MAX];
134 size_t raw_len = 0U;
135 if (!tron_api_hex_to_bytes(raw_hex, raw, sizeof(raw), &raw_len)) {
136 ESP_LOGE(TAG, "raw_data_hex is not valid hex (or too long)");
137 return false;
138 }
139
140 uint8_t txid[CW_HASH_SIZE];
141 size_t txid_len = 0U;
142 if (!tron_api_hex_to_bytes(txid_hex, txid, sizeof(txid), &txid_len) ||
143 (txid_len != CW_HASH_SIZE)) {
144 ESP_LOGE(TAG, "txID is not a 32-byte hex string");
145 return false;
146 }
147
148 /* txID must be the SHA-256 of the bytes we are about to sign — otherwise
149 * the node showed us one transaction and asked us to sign another. */
150 uint8_t digest[CW_HASH_SIZE];
151 if (!crypto.sha256(raw, raw_len, digest)) {
152 ESP_LOGE(TAG, "SHA-256 failed");
153 return false;
154 }
155 if (!CW_Utils::secure_compare(digest, txid, CW_HASH_SIZE)) {
156 ESP_LOGE(TAG, "txID does not match sha256(raw_data) — refusing to sign");
157 return false;
158 }
159
160 const char *reason = tron_verify_transfer(raw, raw_len, owner21, to21, amount);
161 if (reason != NULL) {
162 ESP_LOGE(TAG, "raw_data: %s — refusing to sign", reason);
163 return false;
164 }
165
166 (void)CW_Utils::safe_memcpy(hash, CW_HASH_SIZE, digest, CW_HASH_SIZE);
167 return true;
168}

182static void signing_loop(CryptnoxWallet &wallet, CW_CryptoProvider &crypto,
183 const uint8_t *owner21, const uint8_t *to21)
184{
185 /* CARD_PIN is a string literal ("000000000"); copy into the pin array.
186 * Length comes from the literal itself, so editing the PIN cannot read
187 * past it and send the card a garbage PIN.
188 * ponytail: card_pin is not secure_wipe'd — signing_loop never returns,
189 * so the PIN stays in RAM for the whole run. Read it from a secure source
190 * and wipe after each sign if this goes beyond a demo. */
191 uint8_t card_pin[CW_MAX_PIN_LENGTH] = {};
192 const size_t literal_len = strlen(CARD_PIN);
193 const size_t pin_len = (literal_len < CW_MAX_PIN_LENGTH) ? literal_len
194 : CW_MAX_PIN_LENGTH;
195 (void)CW_Utils::safe_memcpy(card_pin, sizeof(card_pin),
196 reinterpret_cast<const uint8_t *>(CARD_PIN),
197 pin_len);
198
199 while (true) {
200 /* ── 1. Wait for card ──────────────────────────────────── */
201 ESP_LOGI(TAG, "Hold Cryptnox card to reader to sign...");
202
203 CW_SecureSession session;
204 if (!wallet.connect(session)) {
205 vTaskDelay(pdMS_TO_TICKS(100U));
206 continue;
207 }
208
209 /* ── 2. Let the node build the unsigned transfer ────────── */
210 char tx_json[TX_JSON_SIZE];
211 if (!tron_api_create_transfer(TRON_ADDR_FROM, TRON_ADDR_TO,
212 AMOUNT_SUN, tx_json, sizeof(tx_json))) {
213 ESP_LOGE(TAG, "createtransaction failed — retrying in 5 s");
214 wallet.disconnect(session);
215 vTaskDelay(pdMS_TO_TICKS(5000U));
216 continue;
217 }
218
219 /* ── 3. Verify it before the card ever sees it ──────────── */
220 uint8_t hash[CW_HASH_SIZE];
221 if (!verify_transaction(tx_json, crypto, owner21, to21,
222 AMOUNT_SUN, hash)) {
223 wallet.disconnect(session);
224 vTaskDelay(pdMS_TO_TICKS(5000U));
225 continue;
226 }
227
228 ESP_LOGI(TAG, "Verified txID / hash to sign:");
229 ESP_LOG_BUFFER_HEX_LEVEL(TAG, hash, CW_HASH_SIZE, ESP_LOG_INFO);
230
231 /* ── 4. Sign on the card over m/44'/195'/0'/0/0 ─────────── */
232 CW_SignRequest req(session,
233 CW_SIGN_DERIVE_K1,
234 CW_SIGN_SIG_ECDSA_LOW_S,
235 CW_SIGN_WITH_PIN);
236 req.hash = hash;
237 req.hashLength = static_cast<uint8_t>(CW_HASH_SIZE);
238 req.derivePath = CW_TRON_DERIVE_PATH;
239 req.derivePathLength = static_cast<uint8_t>(CW_TRON_PATH_LENGTH);
240 (void)CW_Utils::safe_memcpy(req.pin, sizeof(req.pin),
241 card_pin, CW_MAX_PIN_LENGTH);
242
243 CW_SignResult result = wallet.sign(req);
244 wallet.disconnect(session);
245
246 if (result.errorCode != CW_OK) {
247 ESP_LOGE(TAG, "Sign failed: 0x%02X",
248 static_cast<unsigned int>(result.errorCode));
249 vTaskDelay(pdMS_TO_TICKS(2000U));
250 continue;
251 }
252
253 /* ── 5. Broadcast r||s||v, trying both parities ─────────── *
254 * Retrying cannot double-spend: txID = sha256(raw_data) does not cover
255 * the signature, so both attempts carry the same txID and the node
256 * rejects a duplicate of one already accepted. */
257 uint8_t signature[TRON_SIGNATURE_BYTES];
258 (void)CW_Utils::safe_memcpy(signature, sizeof(signature),
259 result.signature, CW_RAW_SIGNATURE_SIZE);
260
261 bool broadcast = false;
262 for (uint8_t v = 0U; (v < PARITY_CANDIDATES) && !broadcast; v++) {
263 signature[CW_RAW_SIGNATURE_SIZE] = v;
264
265 char resp[512] = {};
266 broadcast = tron_api_broadcast(tx_json, signature, resp, sizeof(resp));
267 if (broadcast) {
268 ESP_LOGI(TAG, "TX broadcast OK (v=%u): %s",
269 static_cast<unsigned int>(v), resp);
270 } else {
271 ESP_LOGW(TAG, "Broadcast rejected with v=%u: %s",
272 static_cast<unsigned int>(v), resp);
273 }
274 }
275
276 if (!broadcast) {
277 ESP_LOGE(TAG, "TX broadcast failed with both parity bits");
278 }
279
280 /* Wait before the next iteration so the transfer can confirm. */
281 vTaskDelay(pdMS_TO_TICKS(15000U));
282 }

296{
297 /* ── NVS (required by WiFi driver) ────────────────────────── */
298 esp_err_t nvs_ret = nvs_flash_init();
299 if ((nvs_ret == ESP_ERR_NVS_NO_FREE_PAGES) ||
300 (nvs_ret == ESP_ERR_NVS_NEW_VERSION_FOUND)) {
301 ESP_ERROR_CHECK(nvs_flash_erase());
302 nvs_ret = nvs_flash_init();
303 }
304 ESP_ERROR_CHECK(nvs_ret);
305
306 /* ── PN532 NFC reader (transport selected at the top of this file) ── */
307 pn532_t nfc = {};
308 pn532_config_t nfc_cfg = {};
309#if SPI_ENABLED
310 spi_bus_config_t buscfg = {};
311 buscfg.mosi_io_num = SPI_MOSI;
312 buscfg.miso_io_num = SPI_MISO;
313 buscfg.sclk_io_num = SPI_SCLK;
314 buscfg.quadwp_io_num = SPI_PIN_UNUSED;
315 buscfg.quadhd_io_num = SPI_PIN_UNUSED;
316 buscfg.max_transfer_sz = SPI_MAX_TRANSFER_SZ;
317 ESP_ERROR_CHECK(spi_bus_initialize(SPI2_HOST, &buscfg, SPI_DMA_CH_AUTO));
318
319 nfc_cfg.transport = PN532_TRANSPORT_SPI;
320 nfc_cfg.spi_host = SPI2_HOST;
321 nfc_cfg.pin_cs = NFC_CS;
322 nfc_cfg.skip_bus_init = true;
323#endif
324
325#if I2C_ENABLED
326 nfc_cfg.transport = PN532_TRANSPORT_I2C;
327 nfc_cfg.i2c_port = PN532_I2C_PORT;
328 nfc_cfg.pin_sda = PN532_SDA;
329 nfc_cfg.pin_scl = PN532_SCL;
330 nfc_cfg.pin_irq = PN532_IRQ;
331 nfc_cfg.pin_rst = PN532_RST;
332 nfc_cfg.i2c_clock_hz = PN532_I2C_HZ;
333#endif
334
335 if (pn532_init(&nfc, &nfc_cfg) != ESP_OK) {
336 ESP_LOGE(TAG, "PN532 init failed");
337 return;
338 }
339
340 /* ── Wallet setup ──────────────────────────────────────────── */
341 ESP32Logger logger;
342 (void)logger.begin(115200UL);
343
344 ESP32CryptoProvider cryptoProvider;
345 ESP32Platform platform;
346 Pn532NfcTransport nfcTransport(&nfc, logger);
347 CryptnoxWallet wallet(nfcTransport, logger, cryptoProvider, platform);
348
349 if (!wallet.begin()) {
350 ESP_LOGE(TAG, "Wallet begin (SAMConfig) failed");
351 return;
352 }
353
354 /* ── Validate the configured addresses once, up front ───────── */
355 uint8_t owner21[CW_TRON_ADDRESS_BYTES];
356 uint8_t to21[CW_TRON_ADDRESS_BYTES];
357 if (!CW_Tron::decodeAddress(TRON_ADDR_FROM, cryptoProvider, owner21)) {
358 ESP_LOGE(TAG, "TRON_ADDR_FROM is not a valid TRON address");
359 return;
360 }
361 if (!CW_Tron::decodeAddress(TRON_ADDR_TO, cryptoProvider, to21)) {
362 ESP_LOGE(TAG, "TRON_ADDR_TO is not a valid TRON address");
363 return;
364 }
365
366 /* ── WiFi + node ───────────────────────────────────────────── */
367 tron_api_init(TRON_NODE_URL, TRON_API_KEY);
368
369 if (!tron_api_wifi_connect(WIFI_SSID, WIFI_PASSWORD)) {
370 ESP_LOGE(TAG, "WiFi connect failed — check config.h credentials");
371 return;
372 }
373
374 ESP_LOGI(TAG, "Ready — will send %" PRIu64 " SUN from %s to %s on each card tap",
375 (uint64_t)AMOUNT_SUN, TRON_ADDR_FROM, TRON_ADDR_TO);
376
377 signing_loop(wallet, cryptoProvider, owner21, to21);
378}

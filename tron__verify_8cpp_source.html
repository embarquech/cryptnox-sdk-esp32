85static bool pb_read_varint(pb_reader_t *r, uint64_t *out)
86{
87 uint64_t value = 0U;
88 uint32_t count = 0U;
89 bool done = false;
90 bool ok = true;
91
92 while ((!done) && ok) {
93 if ((r->pos >= r->len) || (count >= VARINT_MAX_BYTES)) {
94 ok = false;
95 } else {
96 const uint8_t byte = r->buf[r->pos];
97 r->pos++;
98 if ((count == (VARINT_MAX_BYTES - 1U)) &&
99 ((uint32_t)(byte & VARINT_PAYLOAD_MASK) > VARINT_LAST_BYTE_MAX)) {
100 ok = false;
101 } else {
102 value |= ((uint64_t)(byte & VARINT_PAYLOAD_MASK)) << (count * VARINT_BITS_PER_BYTE);
103 done = ((byte & VARINT_CONT_BIT) == 0U);
104 }
105 count++;
106 }
107 }
108 *out = value;
109 return ok;
110}

113static bool pb_skip(pb_reader_t *r, size_t n)
114{
115 bool ok = (n <= (r->len - r->pos));
116 if (ok) {
117 r->pos += n;
118 }
119 return ok;

123static bool pb_next(pb_reader_t *r, pb_field_t *f)
124{
125 uint64_t tag = 0U;
126 bool ok = false;
127
128 (void)memset(f, 0, sizeof(*f));
129 ok = pb_read_varint(r, &tag);
130 if (ok) {
131 f->number = tag >> PB_TAG_SHIFT;
132 f->wire = (uint32_t)(tag & PB_WIRE_MASK);
133 ok = (f->number != 0U);
134 }
135 if (ok) {
136 switch (f->wire) {
137 case PB_WIRE_VARINT:
138 ok = pb_read_varint(r, &f->varint);
139 break;
140 case PB_WIRE_LEN:
141 ok = pb_read_varint(r, &f->varint);
142 if (ok) {
143 ok = (f->varint <= (uint64_t)(r->len - r->pos));
144 }
145 if (ok) {
146 f->data = &r->buf[r->pos];
147 f->data_len = (size_t)f->varint;
148 r->pos += f->data_len;
149 }
150 break;
151 case PB_WIRE_FIXED64:
152 ok = pb_skip(r, PB_FIXED64_BYTES);
153 break;
154 case PB_WIRE_FIXED32:
155 ok = pb_skip(r, PB_FIXED32_BYTES);
156 break;
157 default:
158 ok = false;
159 break;
160 }
161 }
162 return ok;
163}

178static const char *check_transfer(const uint8_t *buf, size_t len,
179 const uint8_t *owner21, const uint8_t *to21,
180 uint64_t amount)
181{
182 pb_reader_t r = { buf, len, 0U };
183 pb_field_t f;
184 bool seen_owner = false;
185 bool seen_to = false;
186 bool seen_amount = false;
187 const char *reason = NULL;
188
189 (void)memset(&f, 0, sizeof(f));
190 while ((reason == NULL) && (r.pos < r.len)) {
191 if (!pb_next(&r, &f)) {
192 reason = ERR_MALFORMED;
193 } else if (pb_is(&f, TRANSFER_FIELD_OWNER, PB_WIRE_LEN) && (!seen_owner)) {
194 seen_owner = true;
195 if (!pb_bytes_equal(&f, owner21, TRON_ADDR_RAW_BYTES)) {
196 reason = ERR_OWNER;
197 }
198 } else if (pb_is(&f, TRANSFER_FIELD_TO, PB_WIRE_LEN) && (!seen_to)) {
199 seen_to = true;
200 if (!pb_bytes_equal(&f, to21, TRON_ADDR_RAW_BYTES)) {
201 reason = ERR_TO;
202 }
203 } else if (pb_is(&f, TRANSFER_FIELD_AMOUNT, PB_WIRE_VARINT) && (!seen_amount)) {
204 seen_amount = true;
205 if (f.varint != amount) {
206 reason = ERR_AMOUNT;
207 }
208 } else {
209 reason = ERR_UNEXPECTED;
210 }
211 }
212
213 if (reason == NULL) {
214 if (!seen_owner) {
215 reason = ERR_OWNER;
216 } else if (!seen_to) {
217 reason = ERR_TO;
218 } else if (!seen_amount) {
219 reason = ERR_AMOUNT;
220 } else {
221 /* all three present and matching */
222 }
223 }
224 return reason;
225}

228static const char *check_any(const uint8_t *buf, size_t len,
229 const uint8_t *owner21, const uint8_t *to21,
230 uint64_t amount)
231{
232 pb_reader_t r = { buf, len, 0U };
233 pb_field_t f;
234 bool seen_url = false;
235 const uint8_t *value = NULL;
236 size_t value_len = 0U;
237 const char *reason = NULL;
238
239 (void)memset(&f, 0, sizeof(f));
240 while ((reason == NULL) && (r.pos < r.len)) {
241 if (!pb_next(&r, &f)) {
242 reason = ERR_MALFORMED;
243 } else if (pb_is(&f, ANY_FIELD_TYPE_URL, PB_WIRE_LEN) && (!seen_url)) {
244 seen_url = true;
245 if (!pb_bytes_equal(&f, (const uint8_t *)TRANSFER_TYPE_URL,
246 sizeof(TRANSFER_TYPE_URL) - 1U)) {
247 reason = ERR_NOT_TRANSFER;
248 }
249 } else if (pb_is(&f, ANY_FIELD_VALUE, PB_WIRE_LEN) && (value == NULL)) {
250 value = f.data;
251 value_len = f.data_len;
252 } else {
253 reason = ERR_UNEXPECTED;
254 }
255 }
256
257 if (reason == NULL) {
258 if (!seen_url) {
259 reason = ERR_NOT_TRANSFER;
260 } else if (value == NULL) {
261 reason = ERR_MALFORMED;
262 } else {
263 reason = check_transfer(value, value_len, owner21, to21, amount);
264 }
265 }
266 return reason;
267}

270static const char *check_contract(const uint8_t *buf, size_t len,
271 const uint8_t *owner21, const uint8_t *to21,
272 uint64_t amount)
273{
274 pb_reader_t r = { buf, len, 0U };
275 pb_field_t f;
276 bool seen_type = false;
277 const uint8_t *param = NULL;
278 size_t param_len = 0U;
279 const char *reason = NULL;
280
281 (void)memset(&f, 0, sizeof(f));
282 while ((reason == NULL) && (r.pos < r.len)) {
283 if (!pb_next(&r, &f)) {
284 reason = ERR_MALFORMED;
285 } else if (pb_is(&f, CONTRACT_FIELD_TYPE, PB_WIRE_VARINT) && (!seen_type)) {
286 seen_type = true;
287 if (f.varint != CONTRACT_TYPE_TRANSFER) {
288 reason = ERR_NOT_TRANSFER;
289 }
290 } else if (pb_is(&f, CONTRACT_FIELD_PARAMETER, PB_WIRE_LEN) && (param == NULL)) {
291 param = f.data;
292 param_len = f.data_len;
293 } else {
294 reason = ERR_UNEXPECTED;
295 }
296 }
297
298 if (reason == NULL) {
299 if (!seen_type) {
300 reason = ERR_NOT_TRANSFER; /* proto3 default 0 = AccountCreateContract */
301 } else if (param == NULL) {
302 reason = ERR_MALFORMED;
303 } else {
304 reason = check_any(param, param_len, owner21, to21, amount);
305 }
306 }
307 return reason;
308}

310const char *tron_verify_transfer(const uint8_t *raw, size_t raw_len,
311 const uint8_t *owner21, const uint8_t *to21,
312 uint64_t amount)
313{
314 pb_reader_t r = { raw, raw_len, 0U };
315 pb_field_t f;
316 const uint8_t *contract = NULL;
317 size_t contract_len = 0U;
318 const char *reason = NULL;
319
320 (void)memset(&f, 0, sizeof(f));
321 if ((raw == NULL) || (owner21 == NULL) || (to21 == NULL)) {
322 reason = ERR_MALFORMED;
323 }
324
325 /* Other raw fields (ref block, expiration, timestamp, memo...) do not
326 * move funds and are only checked for being well-formed. */
327 while ((reason == NULL) && (r.pos < r.len)) {
328 if (!pb_next(&r, &f)) {
329 reason = ERR_MALFORMED;
330 } else if (pb_is(&f, RAW_FIELD_CONTRACT, PB_WIRE_LEN)) {
331 if (contract != NULL) {
332 reason = ERR_CONTRACTS;
333 } else {
334 contract = f.data;
335 contract_len = f.data_len;
336 }
337 } else if (f.number == RAW_FIELD_CONTRACT) {
338 reason = ERR_MALFORMED;
339 } else {
340 /* skipped */
341 }
342 }
343
344 if (reason == NULL) {
345 if (contract == NULL) {
346 reason = ERR_CONTRACTS;
347 } else {
348 reason = check_contract(contract, contract_len, owner21, to21, amount);
349 }
350 }
351 return reason;
352}

371static size_t encode_varint(uint64_t value, uint8_t *out)
372{
373 uint64_t v = value;
374 size_t n = 0U;
375 do {
376 uint8_t byte = (uint8_t)(v & VARINT_PAYLOAD_MASK);
377 v >>= VARINT_BITS_PER_BYTE;
378 if (v != 0U) {
379 byte |= VARINT_CONT_BIT;
380 }
381 out[n] = byte;
382 n++;
383 } while (v != 0U);
384 return n;
385}

388static size_t put_len(uint8_t *out, uint8_t tag, const uint8_t *data, size_t len)
389{
390 size_t n = 0U;
391 out[n] = tag;
392 n++;
393 n += encode_varint((uint64_t)len, &out[n]);
394 (void)memcpy(&out[n], data, len);
395 n += len;
396 return n;
397}

399/* A Transaction.raw for a TRX transfer, laid out per the protobuf schema:
400 * raw: 1 ref_block_bytes | 4 ref_block_hash | 8 expiration
401 * 11 contract { 1 type | 2 parameter: Any { 1 type_url | 2 value } }
402 * 14 timestamp
403 * @p extra is appended to the inner contract value, to model smuggled bytes. */
404static size_t build_sample_raw(uint8_t *out, uint64_t type, const char *type_url,
405 const uint8_t *owner21, const uint8_t *to21,
406 uint64_t amount, const uint8_t *extra, size_t extra_len)
407{
408 uint8_t value[128];
409 uint8_t any[256];
410 uint8_t contract[300];
411 size_t t = 0U;
412 size_t a = 0U;
413 size_t c = 0U;
414 size_t n = 0U;
415
416 t += put_len(&value[t], TAG_OWNER, owner21, TRON_ADDR_RAW_BYTES);
417 t += put_len(&value[t], TAG_TO, to21, TRON_ADDR_RAW_BYTES);
418 value[t] = TAG_AMOUNT;
419 t++;
420 t += encode_varint(amount, &value[t]);
421 if (extra_len > 0U) {
422 (void)memcpy(&value[t], extra, extra_len);
423 t += extra_len;
424 }
425
426 a += put_len(&any[a], 0x0AU, (const uint8_t *)type_url, strlen(type_url));
427 a += put_len(&any[a], 0x12U, value, t);
428
429 contract[c] = 0x08U;
430 c++;
431 c += encode_varint(type, &contract[c]);
432 c += put_len(&contract[c], 0x12U, any, a);
433
434 static const uint8_t REF_BYTES[] = { 0x02U, 0xDBU };
435 static const uint8_t REF_HASH[] = { 0xC0U, 0xC1U, 0xC2U, 0xC3U, 0xC4U, 0xC5U, 0xC6U, 0xC7U };
436 n += put_len(&out[n], 0x0AU, REF_BYTES, sizeof(REF_BYTES));
437 n += put_len(&out[n], 0x22U, REF_HASH, sizeof(REF_HASH));
438 out[n] = 0x40U; /* expiration */
439 n++;
440 n += encode_varint(1739000000000ULL, &out[n]);
441 n += put_len(&out[n], 0x5AU, contract, c);
442 out[n] = 0x70U; /* timestamp */
443 n++;
444 n += encode_varint(1738999000000ULL, &out[n]);
445 return n;
446}

448int main(void)
449{
450 uint8_t owner[TRON_ADDR_RAW_BYTES];
451 uint8_t to[TRON_ADDR_RAW_BYTES];
452 uint8_t other[TRON_ADDR_RAW_BYTES];
453 for (unsigned i = 0U; i < TRON_ADDR_RAW_BYTES; i++) {
454 owner[i] = (uint8_t)((i == 0U) ? 0x41U : (0x10U + i));
455 to[i] = (uint8_t)((i == 0U) ? 0x41U : (0x50U + i));
456 other[i] = (uint8_t)((i == 0U) ? 0x41U : (0x90U + i));
457 }
458
459 uint8_t raw[512];
460 uint8_t extra[64];
461 const uint64_t amount = 1000000ULL; /* 1 TRX */
462 size_t len = build_sample_raw(raw, CONTRACT_TYPE_TRANSFER, TRANSFER_TYPE_URL,
463 owner, to, amount, NULL, 0U);
464
465 /* Matching transaction is accepted. */
466 assert(tron_verify_transfer(raw, len, owner, to, amount) == NULL);
467
468 /* A node that swapped the recipient, sender, or amount is caught. */
469 assert(tron_verify_transfer(raw, len, owner, other, amount) != NULL);
470 assert(tron_verify_transfer(raw, len, other, to, amount) != NULL);
471 assert(tron_verify_transfer(raw, len, owner, to, amount + 1U) != NULL);
472
473 /* Truncated input must not read out of bounds or pass — except the cut
474 * that drops exactly the trailing timestamp, which is a valid message. */
475 const size_t valid_cut = len - 1U - encode_varint(1738999000000ULL, extra);
476 for (size_t cut = 0U; cut < len; cut++) {
477 if (cut != valid_cut) {
478 assert(tron_verify_transfer(raw, cut, owner, to, amount) != NULL);
479 }
480 }
481
482 /* A transaction paying `owner` twice must not pass as owner -> to. */
483 len = build_sample_raw(raw, CONTRACT_TYPE_TRANSFER, TRANSFER_TYPE_URL,
484 owner, owner, amount, NULL, 0U);
485 assert(tron_verify_transfer(raw, len, owner, to, amount) != NULL);
486
487 /* TriggerSmartContract shares tags 1/2/3 (owner, contract, call_value):
488 * same field bytes, plus `12 15 <to>` smuggled after them, is rejected. */
489 size_t e = put_len(extra, TAG_TO, to, TRON_ADDR_RAW_BYTES);
490 len = build_sample_raw(raw, CONTRACT_TYPE_TRIGGER, TRIGGER_TYPE_URL,
491 owner, other, amount, extra, e);
492 assert(tron_verify_transfer(raw, len, owner, to, amount) != NULL);
493 len = build_sample_raw(raw, CONTRACT_TYPE_TRIGGER, TRIGGER_TYPE_URL,
494 owner, to, amount, NULL, 0U);
495 assert(tron_verify_transfer(raw, len, owner, to, amount) != NULL);
496
497 /* Right contract type but wrong Any.type_url is rejected. */
498 len = build_sample_raw(raw, CONTRACT_TYPE_TRANSFER, TRIGGER_TYPE_URL,
499 owner, to, amount, NULL, 0U);
500 assert(tron_verify_transfer(raw, len, owner, to, amount) != NULL);
501
502 /* Wrong real amount with the expected `18 <amount>` embedded after it. */
503 e = 0U;
504 extra[e] = TAG_AMOUNT;
505 e++;
506 e += encode_varint(amount, &extra[e]);
507 len = build_sample_raw(raw, CONTRACT_TYPE_TRANSFER, TRANSFER_TYPE_URL,
508 owner, to, 999U * amount, extra, e);
509 assert(tron_verify_transfer(raw, len, owner, to, amount) != NULL);
510
511 /* An unknown field inside TransferContract is rejected. */
512 extra[0] = 0x28U; /* field 5, VARINT */
513 extra[1] = 0x01U;
514 len = build_sample_raw(raw, CONTRACT_TYPE_TRANSFER, TRANSFER_TYPE_URL,
515 owner, to, amount, extra, 2U);
516 assert(tron_verify_transfer(raw, len, owner, to, amount) != NULL);
517
518 /* A second contract appended to raw is rejected. */
519 len = build_sample_raw(raw, CONTRACT_TYPE_TRANSFER, TRANSFER_TYPE_URL,
520 owner, to, amount, NULL, 0U);
521 uint8_t second[512];
522 const size_t second_len = build_sample_raw(second, CONTRACT_TYPE_TRIGGER, TRIGGER_TYPE_URL,
523 owner, other, amount, NULL, 0U);
524 (void)memcpy(&raw[len], second, second_len);
525 assert(tron_verify_transfer(raw, len + second_len, owner, to, amount) != NULL);
526
527 printf("tron_verify self-check OK\n");
528 return 0;
529}
